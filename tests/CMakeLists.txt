cmake_minimum_required(VERSION 3.16)
project(qiscreen_protocol_test C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(UX "${ROOT}/third_party/UxPlay/lib")
set(PL "${ROOT}/third_party/libplist")
find_package(OpenSSL REQUIRED)
file(GLOB PLIST_C "${PL}/src/*.c")
add_library(plist_test STATIC ${PLIST_C} "${PL}/libcnary/node.c" "${PL}/libcnary/node_list.c")
target_include_directories(plist_test PUBLIC "${PL}/include" PRIVATE "${PL}/src" "${PL}/libcnary/include")
target_compile_definitions(plist_test PRIVATE PACKAGE_VERSION="2.8.0" HAVE_STRNDUP=1 HAVE_MEMMEM=1 HAVE_STRPTIME=1
    HAVE_GMTIME_R=1 HAVE_LOCALTIME_R=1 HAVE_TM_TM_GMTOFF=1 HAVE_TM_TM_ZONE=1)
file(GLOB PROTOCOL_C "${UX}/*.c" "${UX}/playfair/*.c" "${UX}/llhttp/*.c" "${UX}/mdnsd/*.c")
add_library(protocol_test STATIC ${PROTOCOL_C})
target_include_directories(protocol_test PUBLIC "${UX}" PRIVATE "${UX}/playfair" "${UX}/llhttp" "${UX}/mdnsd")
target_compile_definitions(protocol_test PRIVATE PLIST_210 PLIST_230 OPENSSL_API_COMPAT=0x10101000L)
target_compile_options(protocol_test PRIVATE -Wno-deprecated-declarations)
target_link_libraries(protocol_test PUBLIC OpenSSL::Crypto plist_test)
add_executable(protocol_smoke protocol_smoke.c)
target_link_libraries(protocol_smoke PRIVATE protocol_test)
enable_testing()
add_test(NAME rtsp_lifecycle COMMAND protocol_smoke)
set_tests_properties(rtsp_lifecycle PROPERTIES TIMEOUT 20)
add_executable(connection_regressions connection_regressions.cpp)
target_include_directories(connection_regressions PRIVATE "${ROOT}/entry/src/main/cpp")
target_link_libraries(connection_regressions PRIVATE protocol_test)
add_test(NAME latest_lifecycle_intent COMMAND connection_regressions lifecycle)
add_test(NAME cached_endpoint_reuse COMMAND connection_regressions reuse)
add_test(NAME occupied_endpoint_fallback COMMAND connection_regressions fallback)
add_test(NAME idle_stop_latency COMMAND connection_regressions idle-stop)
add_test(NAME protocol_descriptor_cleanup COMMAND connection_regressions resources)
set_tests_properties(idle_stop_latency protocol_descriptor_cleanup PROPERTIES TIMEOUT 20)
set_tests_properties(latest_lifecycle_intent cached_endpoint_reuse occupied_endpoint_fallback PROPERTIES TIMEOUT 20)
add_executable(latency_policy_tests latency_policy_tests.cpp)
target_include_directories(latency_policy_tests PRIVATE "${ROOT}/entry/src/main/cpp")
add_test(NAME decoded_frame_catchup COMMAND latency_policy_tests)
add_executable(tap_tracker_tests tap_tracker_tests.cpp)
target_include_directories(tap_tracker_tests PRIVATE "${ROOT}/entry/src/main/cpp")
add_test(NAME native_surface_controls COMMAND tap_tracker_tests)
add_executable(decoder_queue_tests decoder_queue_tests.cpp)
target_include_directories(decoder_queue_tests PRIVATE "${ROOT}/entry/src/main/cpp")
add_test(NAME compressed_reference_integrity COMMAND decoder_queue_tests)
set_tests_properties(compressed_reference_integrity PROPERTIES TIMEOUT 10)
add_executable(view_gesture_tests view_gesture_tests.cpp)
target_include_directories(view_gesture_tests PRIVATE "${ROOT}/entry/src/main/cpp")
add_test(NAME local_view_zoom COMMAND view_gesture_tests)
add_executable(hevc_config_tests hevc_config_tests.cpp)
target_include_directories(hevc_config_tests PRIVATE "${UX}")
add_test(NAME hevc_parameter_sets COMMAND hevc_config_tests)
add_executable(mirror_lifecycle_tests mirror_lifecycle_tests.cpp)
target_link_libraries(mirror_lifecycle_tests PRIVATE protocol_test)
add_test(NAME mirror_peer_eof_cleanup COMMAND mirror_lifecycle_tests eof)
add_test(NAME mirror_reset_restart COMMAND mirror_lifecycle_tests reset)
set_tests_properties(mirror_peer_eof_cleanup mirror_reset_restart PROPERTIES TIMEOUT 10)
add_executable(runtime_efficiency_tests runtime_efficiency_tests.cpp)
target_include_directories(runtime_efficiency_tests PRIVATE "${ROOT}/entry/src/main/cpp")
add_test(NAME pcm_ring_and_ui_event_priority COMMAND runtime_efficiency_tests)
set_tests_properties(pcm_ring_and_ui_event_priority PROPERTIES TIMEOUT 15)

add_test(NAME mirror_invalid_payload_length COMMAND mirror_lifecycle_tests oversize)
set_tests_properties(mirror_invalid_payload_length PROPERTIES TIMEOUT 10)
